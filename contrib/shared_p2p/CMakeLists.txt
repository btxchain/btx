cmake_minimum_required(VERSION 3.20)
project(btx_shared_p2p_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
get_filename_component(BTX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(name IN ITEMS core broker)
  add_executable(shared_p2p_${name}_test tests/${name}_test.cpp)
  target_include_directories(shared_p2p_${name}_test PRIVATE "${BTX_ROOT}/src")
  target_link_libraries(shared_p2p_${name}_test PRIVATE Threads::Threads)
  add_test(NAME shared_p2p_${name} COMMAND shared_p2p_${name}_test)
  set_tests_properties(shared_p2p_${name} PROPERTIES TIMEOUT 20 SKIP_RETURN_CODE 77)
endforeach()
